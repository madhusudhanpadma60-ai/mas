    e.preventDefault();

    let name = document.getElementById("name").value;
    let email = document.getElementById("email").value;
    let phone = document.getElementById("phone").value;
    let gender = document.getElementById("gender").value;
    let college = document.getElementById("college").value;
    let dept = document.getElementById("dept").value;
    let year = document.getElementById("year").value;
    let event = document.getElementById("event").value;

    let result = document.getElementById("result");

    if (!name || !email || !phone || !gender ||
        !college || !dept || !year || !event) {
        result.innerHTML = "Please fill all fields!";
        result.style.color = "red";
    }
    else if (!email.includes("@") || !email.includes(".")) {
        result.innerHTML = "Invalid Email!";
        result.style.color = "red";
    }
    else if (!/^[0-9]{10}$/.test(phone)) {
        result.innerHTML = "Enter valid 10 digit phone number!";
        result.style.color = "red";
    }
    else {
        result.style.color = "green";
        result.innerHTML =
            "Registration Successful!<br>" +
            "Name: " + name + "<br>" +
            "Email: " + email + "<br>" +
            "Phone: " + phone + "<br>" +
            "Gender: " + gender + "<br>" +
            "College: " + college + "<br>" +
            "Department: " + dept + "<br>" +
            "Year: " + year + "<br>" +
            "Event: " + event;
    }
};

    document.getElementById("result").innerHTML = "";
};

Aim

Requirements

Introduction
Docker is an open-source platform used to develop, build and run applications inside containers.

A container is a lightweight environment that contains an application and its required dependencies.

Docker is used to:
- Create and manage containers.
- Download and manage images.
- Start and stop containers.
- Manage storage volumes.
- Manage Docker networks.
Important Docker Commands
1. Check Docker Version
docker --version


Displays the installed Docker version.

2. Display Docker Information
docker info


Displays complete Docker system information.

3. Download Docker Image
docker pull nginx


Downloads the Nginx image from Docker Hub.

4. Display Docker Images
docker images


Displays all downloaded Docker images.

5. Create and Run Container
docker run -d -p 8080:80 --name mycontainer nginx


Creates and starts an Nginx container.

6. Display Running Containers
docker ps


Displays currently running containers.

7. Display All Containers
docker ps -a


Displays running and stopped containers.

8. Stop Container
docker stop mycontainer


Stops the running container.

9. Start Container
docker start mycontainer


Starts the stopped container.

10. Restart Container
docker restart mycontainer


Restarts the container.

11. Remove Container
docker rm mycontainer


Removes the stopped container.

12. Remove Docker Image
docker rmi nginx


Removes the Docker image if it is not being used by a container.

13. Display Container Logs
docker logs mycontainer


Displays container execution logs.

14. Access Container
docker exec -it mycontainer sh


Opens the terminal inside the container.

15. Copy Files into Container
docker cp index.html mycontainer:/usr/share/nginx/html/


Copies a file into the container.

16. Inspect Container
docker inspect mycontainer


Displays detailed information about the container.

17. Create Docker Volume
docker volume create myvolume


Creates a Docker volume for storing data.

18. Display Docker Volumes
docker volume ls


Displays all Docker volumes.

19. Create Docker Network
docker network create mynetwork


Creates a Docker network.

20. Display Docker Networks
docker network ls


Displays all Docker networks.

21. Remove Unused Docker Resources
docker system prune


Removes unused Docker resources after confirmation.
Procedure
1. Install Docker Desktop.
2. Open Docker Desktop and start Docker Engine.
3. Open Command Prompt.
4. Execute docker --version.
5. Download Nginx using docker pull nginx.
6. View images using docker images.
7. Create a container using docker run.
8. View containers using docker ps.
9. Stop and restart the container.
10. Create volumes and networks.
11. Remove unused containers and images.
Expected Output
Docker version  XX.X.X

REPOSITORY    TAG       IMAGE ID
nginx         latest    xxxxx

CONTAINER ID   IMAGE    STATUS
xxxxx          nginx    Up

http://localhost:8080

1. Node.js Application Using Docker

const http = require('http');

http.createServer((req, res) => {
    res.end("Hello from Node.js Docker Container!");
}).listen(3000);

Step 3: Create Dockerfile
FROM node:18
WORKDIR /app
COPY app.js .
EXPOSE 3000
CMD ["node", "app.js"]


Step 4: Build Docker image
docker build -t nodeapp .

Step 5: Run Docker container
